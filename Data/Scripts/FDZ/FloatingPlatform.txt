private function FloatingPlatform_Type_Linear
	temp0 = object.amplitude.x
	temp0 <<= 8
	object.collisionOffset.x = temp0
	Cos256(temp1, object.angle)
	object.collisionOffset.x *= temp1
	object.collisionOffset.x += object.xpos
	object.collisionOffset.x &= -0x10000
	object.collisionOffset.x -= object.drawPos.x

	temp0 = object.stoodAngle
	temp0 <<= 1
	Sin(temp2, temp0)
	temp2 <<= 9

	temp0 = object.amplitude.y
	temp0 <<= 8
	object.collisionOffset.y = temp0
	Cos256(temp1, object.angle)
	object.collisionOffset.y *= temp1
	object.collisionOffset.y += object.ypos
	if object.hasTension == true
		object.collisionOffset.y += temp2
	end if
	object.collisionOffset.y &= -0x10000
	object.collisionOffset.y -= object.drawPos.y
end function

private function FloatingPlatform_Type_Swing
	temp1 = object.amplitude.y
	temp1++
	temp1 <<= 4
	temp1 -= 8
	object.xpos = object.drawPos.x
	object.ypos = object.drawPos.y

	Cos(object.collisionOffset.x, object.swingAngle)
	object.collisionOffset.x *= temp1
	object.collisionOffset.x <<= 7
	object.collisionOffset.x += temp2
	object.collisionOffset.x &= 0xFFFF0000
	object.drawPos.x = object.collisionOffset.x
	object.collisionOffset.x -= object.xpos

	Sin(object.collisionOffset.y, object.swingAngle)
	object.collisionOffset.y *= temp1
	object.collisionOffset.y <<= 7
	object.collisionOffset.y += temp3
	object.collisionOffset.y &= 0xFFFF0000
	object.drawPos.y = object.collisionOffset.y
	object.collisionOffset.y -= object.ypos
end function

private function FloatingPlatform_Draw_Swing
	// Draw the post
	DrawSprite(0)
	
	// Draw the chains
	temp0 = 0
	temp1 = 16
	while temp0 < object.amplitude.y
		Cos256(temp2, object.swingAngle)
		temp2 *= temp1
		temp2 <<= 7
		temp2 += object.xpos
		Sin256(temp3, object.swingAngle)
		temp3 *= temp1
		temp3 <<= 7
		temp3 += object.ypos
		DrawSpriteXY(1, temp2, temp3)
		temp0++
		temp1 += 16
	loop
	
	// Draw the platform itself
	temp1 -= 8
	Cos256(object.drawPos.x, object.swingAngle)
	object.drawPos.x *= temp1
	object.drawPos.x <<= 7
	object.drawPos.x += object.xpos
	object.drawPos.x &= -0x1 0000
	
	Sin256(object.drawPos.y, object.swingAngle)
	object.drawPos.y *= temp1
	object.drawPos.y <<= 7
	object.drawPos.y += object.ypos
	object.drawPos.y &= -0x1 0000

	DrawSpriteXY(2, object.drawPos.x, object.drawPos.y)
end function

event ObjectUpdate

	if object.stood == true
		if object.canFall == true
			if object.timer == 0
				object.priority = PRIORITY_ACTIVE
				object.timer = 30
			end if
		end if

		if object.stoodAngle < 64
			object.stoodAngle += 4
		end if
	else
		if object.stoodAngle > 0
			object.stoodAngle -= 4
		end if
	end if

	temp0 = oscillation
	temp0 &= 0xFF

	object.angle = 1
	object.angle *= temp0

	switch object.behaviour
	case 0
		CallFunction(FloatingPlatform_Type_Linear)
		break
	case 1
		CallFunction(FloatingPlatform_Type_Circlular)
		break
	case 2
		CallFunction(FloatingPlatform_Type_Swing)
		break
	end switch

	if object.canFall == true
		switch object.state
		case FPLATFORM_IDLE
			if object.timer != 0
				object.timer--
				if object.timer == 0
					object.state++
					object.timer = 32
				end if
			end if
			break
		case FPLATFORM_FALLING_SOLID
			object.timer--
			if object.timer == 0
				object.state++
				foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
					BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, -8, object.hitboxRight, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					if checkResult == true
						player[currentPlayer].ypos += object.yvel
						player[currentPlayer].yvel = object.yvel
						player[currentPlayer].yvel -= 0x3800
						player[currentPlayer].gravity = GRAVITY_AIR
					end if
				next
			end if

		case FPLATFORM_FALLING
			object.ypos += object.yvel
			object.yvel += 0x3800

			if object.outOfBounds == true
				object.ypos = object.centerPos.y
				object.yvel = 0
				object.timer = 0
				object.state++
			end if
			break
			
		case FPLATFORM_OFFSCREEN
			if object.outOfBounds == true
				object.state = FPLATFORM_IDLE
				object.priority = PRIORITY_BOUNDS
			end if
			break
		end switch
	end if

	if object.state < FPLATFORM_FALLING
		temp0 = object.xpos
		temp1 = object.ypos
		object.xpos = object.drawPos.x
		object.ypos = object.drawPos.y

		object.stood = false
	
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, object.hitboxLeft, -16, object.hitboxRight, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.stood = true
				player[currentPlayer].xpos += object.collisionOffset.x
				player[currentPlayer].ypos += object.collisionOffset.y
			end if
		next

		object.xpos = temp0
		object.ypos = temp1
	end if
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			checkResult = object.propertyValue
			break
			
		case 0 // Sprite
			checkResult = object.propertyValue
			checkResult >>= 4
			checkResult &= 0xF
			break

		case 1 // Sprite
			checkResult = object.propertyValue
			checkResult &= 3
			break

		case 2 // hasTension
			GetBit(checkResult, object.propertyValue, 2)
			break
			
		case 3 // canFall
			GetBit(checkResult, object.propertyValue, 3)
			break

		case 4 // hasTension
			checkResult = object.amplitude.x
			break
			
		case 5 // canFall
			checkResult = object.amplitude.y
			break
			
		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			object.propertyValue = editor.variableValue
			break
			
		case 0 // Sprite
			temp0 = editor.variableValue
			temp0 &= 0xF
			temp0 <<= 4
			
			object.propertyValue &= 0x0F
			object.propertyValue |= temp0
			break

		case 1 // Sprite
			temp1 = 3
			Not(temp1)
			object.propertyValue &= temp1

			temp0 = editor.variableValue
			temp0 &= 3
			object.propertyValue |= temp0
			break

		case 2 // hasTension
			CheckNotEqual(editor.variableValue, false)
			SetBit(object.propertyValue, 2, checkResult)
			break

		case 3 // canFall
			CheckNotEqual(editor.variableValue, false)
			SetBit(object.propertyValue, 3, checkResult)
			break

		case 4 // amplitude.x
			object.amplitude.x = editor.variableValue
			break

		case 5 // amplitude.x
			object.amplitude.y = editor.variableValue
			break
			
		end switch
	end if
end event

event RSDKDraw
	object.frame = object.propertyValue
	object.frame &= 0x70
	object.frame >>= 4

	DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)

	if editor.showGizmos == true
		editor.drawingOverlay = true

		object.drawPos.x = object.xpos
		temp0 = object.amplitude.x
		temp0 <<= 16
		object.drawPos.x += temp0

		object.drawPos.y = object.ypos
		temp0 = object.amplitude.y
		temp0 <<= 16
		object.drawPos.y += temp0
		object.inkEffect = INK_BLEND
		DrawSpriteFX(object.frame, FX_INK, object.drawPos.x, object.drawPos.y)
		object.inkEffect = INK_NONE

		CheckNotEqual(object.amplitude.x, 0)
		temp0 = checkResult
		CheckNotEqual(object.amplitude.y, 0)
		temp0 |= checkResult
		

		//if object.amplitude.x != 0
			if temp0 == true
				DrawArrow(object.xpos, object.ypos, object.drawPos.x, object.drawPos.y, 0x00, 0xFF, 0x00)
			else	
				DrawLine(object.xpos, object.ypos, object.drawPos.x, object.drawPos.y, 0x00, 0xFF, 0x00)
			end if
		//else	
		//	DrawLine(object.xpos, object.ypos, object.drawPos.x, object.drawPos.y, 0x00, 0xFF, 0x00)
		//end if
		object.drawPos.x = object.xpos
		temp0 = object.amplitude.x
		temp0 <<= 16
		object.drawPos.x -= temp0

		object.drawPos.y = object.ypos
		temp0 = object.amplitude.y
		temp0 <<= 16
		object.drawPos.y -= temp0

		object.inkEffect = INK_BLEND
		DrawSpriteFX(object.frame, FX_INK, object.drawPos.x, object.drawPos.y)
		object.inkEffect = INK_NONE
		DrawLine(object.xpos, object.ypos, object.drawPos.x, object.drawPos.y, 0x00, 0xFF, 0x00)

		GetBit(checkResult, object.propertyValue, 3)
		if checkResult != false
			temp0 = object.ypos

			temp1 = object.ypos
			temp1 += 0x60000
			
			temp2 = object.ypos
			temp2 += 0x3E0000
			while temp0 < temp2
				DrawLine(object.xpos, temp0, object.xpos, temp1, 0x00, 0xFF, 0x00)
				
				temp0 += 0x8 0000
				temp1 += 0x8 0000
			loop
		end if
		editor.drawingOverlay = false
	end if
	
	
end event
