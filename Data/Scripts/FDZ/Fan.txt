event ObjectUpdate
	object.priority = PRIORITY_ACTIVE

	temp0 = oscillation
	temp0 <<= 2
	
	GetTableValue(object.range, 10, StageSetup_oscillationTable)
	object.range >>= 8
	object.range -= 112
	object.value4 = object.range
	object.value4 <<= 16
	object.value4 += object.ypos
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, object.range, 16, 32, currentPlayer, 0, 0, 0, 0)
		if checkResult == true
			player[currentPlayer].animation 		= ANI_TWIRL
			player[currentPlayer].animationSpeed 	= 0x50
			player[currentPlayer].scrollTracking 	= true
			player[currentPlayer].gravity 			= GRAVITY_AIR
			player[currentPlayer].state				= Player_State_Air
			player[currentPlayer].tileCollisions 	= true
			player[currentPlayer].speed 			= player[currentPlayer].xvel
			player[currentPlayer].timer 			= 0
			player[currentPlayer].yvel 				= 0
			temp1 = object.value4
			temp1 -= player[currentPlayer].ypos
			temp1 >>= 4
			player[currentPlayer].ypos += temp1
		end if
	next
end event


event ObjectDraw
    object.animationTimer++
    object.animationTimer &= 15
    object.frame = object.animationTimer
    object.frame >>= 2

    temp0 = object.frame
    temp0++
	DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
end event


event ObjectStartup
    
    foreach (TypeName[Fan], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].subType = object[arrayPos0].propertyValue
		object[arrayPos0].subType &= 0xF
        object[arrayPos0].priority = PRIORITY_ALWAYS
    next

	LoadSpriteSheet("FDZ/Objects.gif")
	
	SpriteFrame(-6, -8, 12, 24, 1, 316)		// #0 - Base
    SpriteFrame(-16, -3, 32, 11, 14, 316)	// #1 - Blades 1
    SpriteFrame(-13, -3, 27, 11, 14, 328)	// #2 - Blades 2
	SpriteFrame(-10, -3, 20, 11, 47, 316)	// #3 - Blades 3
	SpriteFrame(-13, -3, 27, 11, 42, 328)   // #4 - Blades 4
end event

event RSDKEdit
	if editor.returnVariable == true
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			checkResult = object.propertyValue
			break
		case 0 // Sprite
			checkResult = object.propertyValue
			break
		end switch
	else
		switch editor.variableID
		case EDIT_VAR_PROPVAL // property value
			object.propertyValue = editor.variableValue
			break
		case 0 // Sprite
			object.propertyValue = editor.variableValue
			break
		end switch
	end if
end event

event RSDKDraw
	DrawSprite(0)
	DrawSprite(1)

	if editor.showGizmos == true
		editor.drawingOverlay = true
		
		// The hitbox oscillates a bit, but it's only by a few pixels, so this is okay
		temp0 = 16; temp1 = 112; temp2 = 16; temp3 = 32;
		CallFunction(EditorHelpers_DrawHitbox)
		
		editor.drawingOverlay = false
	end if
end event
