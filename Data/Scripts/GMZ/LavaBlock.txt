event ObjectUpdate
	temp0 = object.size
	temp0 <<= 3
	temp1 = temp0
	FlipSign(temp1)

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if player[currentPlayer].state != Player_State_LedgePullUp
			BoxCollisionTest(C_SOLID, object.entityPos, temp1, -8, temp0, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_FireHit)
			end if
		end if
	next
	
end event

		DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
	end if
end event

end event
